function doPost(e) { try { var accion = e.parameter.accion; var datos = JSON.parse(e.parameter.datos || '{}'); var ss = SpreadsheetApp.openById(datos.sheetId); if (accion === 'exportar') { var data = JSON.parse(datos.data); var cfg = JSON.parse(datos.cfg || '{}'); guardarMovimientos(ss, data.movimientos || []); guardarDeudas(ss, data.deudas || []); guardarAhorros(ss, data.ahorros || []); guardarMetas(ss, data.metas || []); guardarCfg(ss, cfg); return ContentService.createTextOutput(JSON.stringify({ok: true, msg: 'Exportado'})) .setMimeType(ContentService.MimeType.JSON); } if (accion === 'importar') { var movs = leerMovimientos(ss); var deudas = leerDeudas(ss); var ahorros = leerAhorros(ss); var metas = leerMetas(ss); return ContentService.createTextOutput(JSON.stringify({ ok: true, data: JSON.stringify({ movimientos: movs, deudas: deudas, ahorros: ahorros, metas: metas }) })).setMimeType(ContentService.MimeType.JSON); } return ContentService.createTextOutput(JSON.stringify({ok: false, error: 'Accion no valida: ' + accion})) .setMimeType(ContentService.MimeType.JSON); } catch(e) { return ContentService.createTextOutput(JSON.stringify({ok: false, error: e.message})) .setMimeType(ContentService.MimeType.JSON); } } function guardarMovimientos(ss, items) { var h = ss.getSheetByName('Movimientos') || ss.insertSheet('Movimientos'); h.clear(); var enc = [['id','fecha','tipo','cat','monto','persona','desc','fijo']]; if (!items.length) { h.getRange(1, 1, enc.length, enc[0].length).setValues(enc); return; } var rows = items.map(function(m) { return [m.id, m.fecha, m.tipo, m.cat, m.monto, m.persona, m.desc || '', m.fijo ? 1 : 0]; }); h.getRange(1, 1, enc.length, enc[0].length).setValues(enc); h.getRange(2, 1, rows.length, rows[0].length).setValues(rows); } function guardarDeudas(ss, items) { var h = ss.getSheetByName('Deudas') || ss.insertSheet('Deudas'); h.clear(); var enc = [['id','tipo','persona','acreedor','total','cuota','desc','vence','pagado']]; if (!items.length) { h.getRange(1, 1, enc.length, enc[0].length).setValues(enc); return; } var rows = items.map(function(d) { return [d.id, d.tipo||'debt', d.persona, d.acreedor, d.total, d.cuota||0, d.desc||'', d.vence||'', d.pagado||0]; }); h.getRange(1, 1, enc.length, enc[0].length).setValues(enc); h.getRange(2, 1, rows.length, rows[0].length).setValues(rows); } function guardarAhorros(ss, items) { var h = ss.getSheetByName('Ahorros') || ss.insertSheet('Ahorros'); h.clear(); var enc = [['id','persona','monto','tipo','desc','fecha']]; if (!items.length) { h.getRange(1, 1, enc.length, enc[0].length).setValues(enc); return; } var rows = items.map(function(a) { return [a.id, a.persona, a.monto, a.tipo, a.desc||'', a.fecha]; }); h.getRange(1, 1, enc.length, enc[0].length).setValues(enc); h.getRange(2, 1, rows.length, rows[0].length).setValues(rows); } function guardarMetas(ss, items) { var h = ss.getSheetByName('Metas') || ss.insertSheet('Metas'); h.clear(); var enc = [['id','nombre','objetivo','actual','persona','fecha','emoji']]; if (!items.length) { h.getRange(1, 1, enc.length, enc[0].length).setValues(enc); return; } var rows = items.map(function(m) { return [m.id, m.nombre, m.objetivo, m.actual||0, m.persona, m.fecha||'', m.emoji||'🎯']; }); h.getRange(1, 1, enc.length, enc[0].length).setValues(enc); h.getRange(2, 1, rows.length, rows[0].length).setValues(rows); } function guardarCfg(ss, cfg) { var h = ss.getSheetByName('Cfg') || ss.insertSheet('Cfg'); h.clear(); var enc = [['clave','valor']]; var keys = Object.keys(cfg); var rows = keys.map(function(k) { return [k, JSON.stringify(cfg[k])]; }); if (!rows.length) { h.getRange(1, 1, enc.length, enc[0].length).setValues(enc); return; } h.getRange(1, 1, enc.length, enc[0].length).setValues(enc); h.getRange(2, 1, rows.length, rows[0].length).setValues(rows); } function leerMovimientos(ss) { var h = ss.getSheetByName('Movimientos'); if (!h) return []; var data = h.getDataRange().getValues(); if (data.length < 2) return []; return data.slice(1).map(function(r) { return {id: r[0], fecha: r[1], tipo: r[2], cat: r[3], monto: r[4], persona: r[5], desc: r[6], fijo: !!r[7]}; }).filter(function(m) { return m.id; }); } function leerDeudas(ss) { var h = ss.getSheetByName('Deudas'); if (!h) return []; var data = h.getDataRange().getValues(); if (data.length < 2) return []; return data.slice(1).map(function(r) { return {id: r[0], tipo: r[1]||'debt', persona: r[2], acreedor: r[3], total: r[4], cuota: r[5]||0, desc: r[6]||'', vence: r[7]||'', pagado: r[8]||0}; }).filter(function(d) { return d.id; }); } function leerAhorros(ss) { var h = ss.getSheetByName('Ahorros'); if (!h) return []; var data = h.getDataRange().getValues(); if (data.length < 2) return []; return data.slice(1).map(function(a) { return {id: a[0], persona: a[1], monto: a[2], tipo: a[3], desc: a[4]||'', fecha: a[5]}; }).filter(function(a) { return a.id; }); } function leerMetas(ss) { var h = ss.getSheetByName('Metas'); if (!h) return []; var data = h.getDataRange().getValues(); if (data.length < 2) return []; return data.slice(1).map(function(m) { return {id: m[0], nombre: m[1], objetivo: m[2], actual: m[3]||0, persona: m[4], fecha: m[5]||'', emoji: m[6]||'🎯'}; }).filter(function(m) { return m.id; }); } function doGet() { return ContentService.createTextOutput( 'K&R Google Sheets Sync — Usa POST con accion, datos y sheetId' ).setMimeType(ContentService.MimeType.TEXT); }
